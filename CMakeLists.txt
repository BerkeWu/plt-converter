cmake_minimum_required(VERSION 3.16)
project(plt_converter VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)   # source files contain Chinese literals
endif()

# Qt-independent core (parser, tiling, naming, dev PDF writer)
add_library(pltcore core/plt.cpp core/pltedit.cpp core/layout.cpp core/pdf_min.cpp core/docops.cpp)
target_include_directories(pltcore PUBLIC core)

add_executable(plt2pdf tools/plt2pdf.cpp)
target_link_libraries(plt2pdf pltcore)
if(MSVC)
  target_sources(plt2pdf PRIVATE packaging/utf8.manifest)
endif()
if(WIN32)
  target_link_libraries(plt2pdf shell32)
endif()

add_executable(test_core tests/test_core.cpp)
target_link_libraries(test_core pltcore)
enable_testing()
add_test(NAME core COMMAND test_core ${CMAKE_SOURCE_DIR}/tests/data)

# Watermark module (needs QPDF; skipped when the library is not installed)
set(PLT_HAVE_QPDF OFF)
find_package(qpdf CONFIG QUIET)            # vcpkg / official release package: target qpdf::libqpdf
if(TARGET qpdf::libqpdf)
  set(PLT_QPDF_TARGET qpdf::libqpdf)
  set(PLT_HAVE_QPDF ON)
else()
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(QPDF QUIET IMPORTED_TARGET libqpdf)
    if(QPDF_FOUND)
      set(PLT_QPDF_TARGET PkgConfig::QPDF)
      set(PLT_HAVE_QPDF ON)
    endif()
  endif()
endif()
if(PLT_HAVE_QPDF)
  add_library(pltwm core/wm/wm.cpp)
  target_include_directories(pltwm PUBLIC core)
  target_link_libraries(pltwm PUBLIC ${PLT_QPDF_TARGET})
  target_compile_features(pltwm PUBLIC cxx_std_20)   # QPDF 12 headers need C++20; the rest of the project stays on C++17
else()
  message(STATUS "QPDF not found: watermark module (pdfwm, GUI mode 水印处理) is not built")
endif()

# Qt6 exporter smoke test (QPdfWriter, CJK font embedding)
find_package(Qt6 COMPONENTS Core Gui Widgets Concurrent QUIET)
if(Qt6_FOUND)
  set(CMAKE_AUTOMOC ON)
  add_executable(test_exporter tests/test_exporter.cpp qt/PdfExporterQt.cpp)
  target_link_libraries(test_exporter pltcore Qt6::Gui)
endif()

if(Qt6_FOUND)
  find_package(Qt6 COMPONENTS Test QUIET)
  add_library(pltgui STATIC qt/BatchCore.cpp qt/MainWindowBatch.cpp qt/BatchWidgets.h qt/MainWindow.cpp qt/CanvasWidget.cpp qt/PageRender.cpp qt/PdfExporterQt.cpp)
  target_include_directories(pltgui PUBLIC qt)
  target_link_libraries(pltgui PUBLIC pltcore Qt6::Widgets Qt6::Concurrent)
  if(PLT_HAVE_QPDF)
    target_sources(pltgui PRIVATE qt/WatermarkPanel.cpp qt/DrawView.cpp)
    target_link_libraries(pltgui PUBLIC pltwm)
    target_compile_definitions(pltgui PUBLIC PLT_WITH_WM)
  endif()
  add_executable(plt_gui qt/main.cpp)
  target_link_libraries(plt_gui pltgui)
  set_target_properties(plt_gui PROPERTIES WIN32_EXECUTABLE ON)
  if(MSVC)
    target_sources(plt_gui PRIVATE packaging/utf8.manifest)
  endif()
  if(Qt6Test_FOUND)
    add_executable(test_gui tests/test_gui.cpp)
    target_link_libraries(test_gui pltgui Qt6::Test)
    add_executable(test_batch tests/test_batch.cpp)
    target_link_libraries(test_batch pltgui Qt6::Test)
    add_test(NAME batch COMMAND test_batch)
    set_tests_properties(batch PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 300)
    if(PLT_HAVE_QPDF)
      add_executable(test_wmgui tests/test_wmgui.cpp)
      target_link_libraries(test_wmgui pltgui Qt6::Test)
      add_test(NAME wmgui COMMAND test_wmgui)
      set_tests_properties(wmgui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;WM_DATA=${CMAKE_SOURCE_DIR}/tests/data_wm" TIMEOUT 600)
    endif()
    add_test(NAME gui COMMAND test_gui ${CMAKE_SOURCE_DIR}/tests/data)
    set_tests_properties(gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
endif()

if(PLT_HAVE_QPDF)
  add_executable(pdfwm tools/pdfwm.cpp)
  target_link_libraries(pdfwm pltwm)
  if(MSVC)
    target_sources(pdfwm PRIVATE packaging/utf8.manifest)
  endif()
  add_executable(test_wm tests/test_wm.cpp)
  target_link_libraries(test_wm pltwm)
  add_test(NAME wm COMMAND test_wm ${CMAKE_SOURCE_DIR}/tests/data_wm)
endif()
